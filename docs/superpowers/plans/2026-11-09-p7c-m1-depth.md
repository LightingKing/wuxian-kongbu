## 设计要点

- **判断力**：`Battle.intel`（0~2）由远征按队伍里最高的智力给出（≥160 → 2，≥110 → 1，否则 0）。0 只知道敌人要动，1 看到技能名，2 看到目标。M1 有詹岚（130），所以默认 1。
- **弹药**：新增 `pistol_shot_limited`（同一把手枪，`cost: {"ammo": 1}`），只给郑吒；张杰的沙漠之鹰仍是无限子弹（原著兑换）。郑吒开局 12 发，车厢/激光通道/走廊/下水道能搜到弹药。没子弹时手枪不在合法行动里。
- **感染的代价**：丧尸咬中会带 `infection` 状态，战斗里 5 回合不治就当场变丧尸；带出战斗的人 60 分钟后转化，必须用解毒剂。界面把"感染"标在成员上，并给出"给谁用解毒剂"的按钮。
- **搜索的风险**：地点可以配 `search_encounter`（概率 + 敌人 + 一句提示）。搜索照样拿到东西、照样花时间，但可能把敌人引过来。控制室、走廊、冷冻仓库、下水道都有风险；车厢和激光通道是安全搜刮点。
- **路线收益**：车厢（手榴弹 + 6 发）、激光通道（4 发）、控制室（喷雾 2 + 解毒剂 1，20 分钟）、走廊（6 发）、冷冻仓库（解毒剂 2）、下水道（4 发）。贪搜要拿命换时间。
- **关键词说服**：`persuasion.gd` 把"选了几个对的关键词"折算成 0~3 级；智力 ≥140 的人多一档。逼火焰女皇必须先从线索推理出"她控制着激光"（`requires_fact`），否则开不了口；界面给"推理（10 分钟）"和"硬扛（放弃说服）"两条路，硬扛就是爬行者战。
- **精英战的代价**：两支爬行者不是杂兵。测试固定一条"只点第一个按钮"的路线，断言至少死一个人——这是对"过于简单"的回归保护。
- **界面**：战斗卡显示弹药，时间轴下面显示敌方意图，标题显示目标；地点按钮标"·险"，搜索按钮在有风险的地点变"搜索（有风险）"；说服面板是关键词开关 + 开口/硬扛；成员下面是可用的解毒剂/喷雾按钮。

**不在本计划范围**：真正的 AI 对话（关键词仍由离线裁决，接口留给 `ai/intents.gd`）；守门/撑时间的战斗目标（引擎支持 `survive_until`，但当前数值下敌人死得太快，等平衡时再用）；主神空间界面；美术与音效。

```gdscript
extends RefCounted
## 说服的离线裁决（设计规格 8、9.5.1）：玩家从对方给的关键词里挑话说，匹配得越多越有说服力。
## 需要先推理出某个事实（requires_fact）才开得了口；没有推论时返回 -1，由远征按失败处理。
## 以后接 AI 对话层时，仍然调用 resolve_persuasion(level)，这里只是把"说什么"折算成 0~3 级。

## 返回 0~3；-1 表示"你还没有把握，说不出口"。
static func level(check: Dictionary, keywords: Array, thinker: RefCounted, intel: RefCounted) -> int:
	var fact := String(check.get("requires_fact", ""))
	if fact != "":
		var status: String = intel.status(fact)
		if status != "inferred" and status != "confirmed":
			return -1
	var known: Array = check.get("keywords", [])
	var matched := 0
	for k: String in keywords:
		if known.has(k):
			matched += 1
	var lvl := mini(matched, 3)
	if lvl > 0 and float(thinker.attributes.total("intelligence")) >= 140.0:
		lvl = mini(lvl + 1, 3)
	return lvl
```

func setup(units: Array[RefCounted], options: Dictionary, random: RefCounted) -> Array[Dictionary]:
	rng = random
	field = Battlefield.new(int(options.get("row_capacity", -1)))
	ctb = Ctb.new()
	dark = bool(options.get("dark", false))
	allow_retreat = bool(options.get("allow_retreat", false))
	survive_until = float(options.get("survive_until", -1.0))
	intel = int(options.get("intel", 0))
	max_actions = int(BattleData.rules()["max_actions"])
	var events: Array[Dictionary] = []
	var placed: Array[String] = []
	for u in units:
		if field.add(u):
			ctb.init_unit(u)
			placed.append(u.id)
		else:
			events.append({"type": "placement_failed", "target": u.id})
	events.append({"type": "battle_start", "units": placed})
	var aura := 0.0
	for e: RefCounted in field.side_units("enemy"):
		aura += e.fear_aura
	if aura > 0.0:
		for a: RefCounted in field.side_units("ally"):
			_add_fear(a, aura, "enemy_appear", events)
	for u in field.units:
		if u.controller == "ai":
			_plan(u)
	_advance(events)
	return _emit(events)


## 玩家下令。不合法的行动会被拒绝，战斗状态不变。
func act(action: Dictionary) -> Array[Dictionary]:
	var events: Array[Dictionary] = []
	if outcome != "ongoing" or current == null or not Targeting.is_legal(field, current, action, allow_retreat):
		events.append({"type": "invalid_action", "action": action})
		return events
	_perform(current, action, events)
	_advance(events)
	return _emit(events)


func legal_actions() -> Array[Dictionary]:
	if outcome != "ongoing" or current == null:
		return []
	return Targeting.legal_actions(field, current, allow_retreat)


## 界面上方的时间轴：接下来 count 次行动的单位 id。

	return ctb.preview(field.units, count)


func intent_view(unit_id: String, intel: int) -> Dictionary:
	return UtilityAi.intent_view(field.get_unit(unit_id), intel)


func _emit(events: Array[Dictionary]) -> Array[Dictionary]:
	history.append_array(events)
	return events


## 推进时间轴，让 AI 和失控的单位行动，直到轮到玩家或战斗结束。
func _advance(events: Array[Dictionary]) -> void:
	while true:
		if _check_end(events):
			return
		if action_count >= max_actions:
			_finish("timeout", events)
			return
		var u: RefCounted = ctb.next_actor(field.units)
		if survive_until >= 0.0 and u.next_time >= survive_until:
			_finish("survived", events)
			return
		events.append({"type": "turn_start", "unit": u.id, "time": u.next_time})
		events.append_array(Statuses.start_of_turn(u))
		_after_hp_loss(u, events)
		if not u.in_play():
			continue
		if dark and u.side == "ally":
			_add_fear(u, float(BattleData.rules()["fear"]["darkness_per_turn"]), "darkness", events)
		if Statuses.has_flag(u, "skip_turn"):
			events.append({"type": "turn_skipped", "unit": u.id})
			_end_turn(u, 1.0, "")
			continue
		if Statuses.has_flag(u, "panic"):
			_panic_action(u, events)
			continue
		if u.controller == "player":
			current = u
			return
		var action: Dictionary = u.intent
		if action.is_empty() or not Targeting.is_legal(field, u, action, false):
			action = UtilityAi.choose(field, u, false, dark)
		_perform(u, action, events)


func _perform(u: RefCounted, action: Dictionary, events: Array[Dictionary]) -> void:
	current = null
	events.append({"type": "action", "unit": u.id, "action": action})
	var delays: Dictionary = BattleData.rules()["delay_mult"]
	var mult := 1.0
	var used := ""
	match String(action["type"]):
		"skill":
			used = String(action["skill"])
			mult = _use_skill(u, action, events)
		"move":
			field.move(u, int(action["row"]))
			events.append({"type": "moved", "unit": u.id, "row": u.row})
			mult = float(delays["move"])
		"defend":
			events.append(Statuses.apply(u, "defending"))
			mult = float(delays["defend"])
		"gene_lock":
			u.aftereffect = true
			events.append(Statuses.apply(u, "gene_lock"))
			mult = float(delays["gene_lock"])
		"retreat":
			_finish("retreat", events)
	_end_turn(u, mult, used)


## 行动结束：排下一次行动；冷却减 1（本回合刚用的技能不减）；AI 单位定好下一步意图。
func _end_turn(u: RefCounted, mult: float, used_skill: String) -> void:
	ctb.advance(u, mult)
	action_count += 1
	for k: String in u.cooldowns.keys():
		if k != used_skill:
			u.cooldowns[k] = maxi(0, int(u.cooldowns[k]) - 1)
	if u.controller == "ai" and u.in_play() and outcome == "ongoing":
		_plan(u)


func _plan(u: RefCounted) -> void:
	u.intent = UtilityAi.choose(field, u, false, dark)


func _use_skill(u: RefCounted, action: Dictionary, events: Array[Dictionary]) -> float:
	var skill_id := String(action["skill"])
	var skill := BattleData.skill(skill_id)
	var cost: Dictionary = skill.get("cost", {})
	for k: String in cost:
		u.resources[k] = float(u.resources.get(k, 0.0)) - float(cost[k])
	if skill.has("cooldown"):
		u.cooldowns[skill_id] = int(skill["cooldown"])
	for tid: String in action["targets"]:
		var t: RefCounted = field.get_unit(tid)
		if t != null and t.in_play():
			_apply_skill(u, t, skill_id, events)
	return float(skill.get("delay_mult", 1.0))


## 一个技能作用到一个目标：命中 → 伤害 → 治疗 → 净化 → 恐惧 → 附加状态。
func _apply_skill(u: RefCounted, t: RefCounted, skill_id: String, events: Array[Dictionary]) -> void:
	var skill := BattleData.skill(skill_id)
	if skill.has("accuracy") and rng.randf() >= CombatMath.hit_chance(u, t, skill, dark):
		events.append({"type": "miss", "source": u.id, "target": t.id, "skill": skill_id})
		return
	if float(skill.get("power", 0.0)) > 0.0:
		_deal_damage(u, t, CombatMath.damage(u, t, skill, rng.randf()), events)
		if not t.in_play():
			return
	if skill.has("heal"):
		var before: float = t.hp
		t.hp = minf(t.hp + float(skill["heal"]), t.hp_max)
		events.append({"type": "heal", "source": u.id, "target": t.id, "amount": t.hp - before, "hp": t.hp})
	for c: String in skill.get("cleanse", []):
		if Statuses.remove(t, c):
			events.append({"type": "status_removed", "target": t.id, "status": c})
	if skill.has("fear"):
		_add_fear(t, float(skill["fear"]), "skill", events)
	for a: Dictionary in skill.get("applies", []):
		if rng.randf() < float(a["chance"]):
			events.append(Statuses.apply(t, String(a["status"])))


func _deal_damage(src: RefCounted, t: RefCounted, amount: float, events: Array[Dictionary]) -> void:
	t.hp -= amount
	events.append({"type": "damage", "source": src.id, "target": t.id, "amount": amount, "hp": t.hp})
	if t.hp > 0.0 and amount >= float(BattleData.rules()["fear"]["heavy_wound_ratio"]) * t.hp_max and not Statuses.has_flag(t, "no_pain"):
		_add_fear(t, float(BattleData.rules()["fear"]["heavy_wound"]), "heavy_wound", events)
	_after_hp_loss(t, events)


## 掉血之后：死了就处理死亡（含复活道具），没死就看要不要做低血觉醒检定。
func _after_hp_loss(t: RefCounted, events: Array[Dictionary]) -> void:
	if t.fled:
		return
	if t.turned or t.hp <= 0.0:
		if t.alive or t.turned:
			_on_death(t, events)
		return
	if t.reincarnator and not t.low_hp_checked and t.hp_ratio() < float(BattleData.rules()["gene_lock"]["low_hp_ratio"]):
		events.append_array(Fear.check_low_hp(t, rng.randf()))


func _on_death(t: RefCounted, events: Array[Dictionary]) -> void:
	if t.revive_charges > 0 and not t.turned:
		t.revive_charges -= 1
		t.resources["rebirth_cross"] = t.revive_charges
		t.hp = roundf(t.hp_max * float(BattleData.rules()["revive_hp_ratio"]))
		events.append({"type": "revived", "target": t.id, "hp": t.hp})
		return
	t.alive = false
	events.append({"type": "death", "target": t.id, "turned": t.turned})
	for ally: RefCounted in field.side_units(t.side):
		_add_fear(ally, float(BattleData.rules()["fear"]["ally_death"]), "ally_death", events)


func _add_fear(u: RefCounted, amount: float, reason: String, events: Array[Dictionary]) -> void:
	var delta := Fear.gain(u, amount)
	if delta != 0.0:
		events.append({"type": "fear_changed", "target": u.id, "amount": delta, "fear": u.fear, "reason": reason})
	if u.fear >= Fear.max_fear() and not u.fear_checked:
		events.append_array(Fear.check_full_fear(u, rng.randf()))


## 恐慌（设计规格 3.5）：不听指挥，发呆、往后逃（已在后排就逃离战场），或者对随便一个人出手（可能误伤队友）。
func _panic_action(u: RefCounted, events: Array[Dictionary]) -> void:
	var p: Dictionary = BattleData.rules()["panic"]
	var r: float = rng.randf()
	if r < float(p["freeze"]):
		events.append({"type": "panic_freeze", "unit": u.id})
	elif r < float(p["freeze"]) + float(p["flee"]):
		if field.move(u, u.row + 1):
			events.append({"type": "panic_flee", "unit": u.id, "row": u.row})
		else:
			u.fled = true
			events.append({"type": "fled", "unit": u.id})
	else:
		var others: Array[RefCounted] = []
		for o in field.units:
			if o != u and o.in_play():
				others.append(o)
		var skill_id := "punch"
		for s in u.skills:
			if float(BattleData.skill(s).get("power", 0.0)) > 0.0:
				skill_id = s
				break
		if not others.is_empty():
			var t: RefCounted = others[rng.randi_range(0, others.size() - 1)]
			events.append({"type": "panic_misfire", "unit": u.id, "target": t.id, "skill": skill_id})
			_apply_skill(u, t, skill_id, events)
	_end_turn(u, float(BattleData.rules()["delay_mult"]["panic"]), "")


func _check_end(events: Array[Dictionary]) -> bool:
	if outcome != "ongoing":
		return true
	if field.side_units("enemy").is_empty():
		_finish("victory", events)
		return true
	if field.side_units("ally").is_empty():
		_finish("defeat", events)
		return true
	return false


func _finish(result: String, events: Array[Dictionary]) -> void:
	if outcome != "ongoing":
		return
	outcome = result
	current = null
	events.append({"type": "battle_end", "outcome": result})

func report_battle(b: RefCounted) -> Array[Dictionary]:
	var events: Array[Dictionary] = []
	if st.pending_battle.is_empty():
		events.append({"type": "invalid_action", "reason": "no_battle"})
		return _emit(events)
	var sid := String(st.pending_battle["squad"])
	var carrier := String(st.pending_battle.get("carrier", ""))
	for u: RefCounted in b.field.units:
		if u.side != "ally" or not st.characters.has(u.character_id):
			continue
		var ch: RefCounted = st.characters[u.character_id]
		UnitFactory.write_back(u, ch)
		_settle_items(u, ch, sid if u.character_id == carrier else "")
		if u.alive and Statuses.has(u, "infection") and not st.infected.has(ch.id):
			st.infected[ch.id] = 0
			events.append({"type": "infected", "char": ch.id})
	var summary := BattleReport.summarize(b.history, b.field.units)
	for k: Dictionary in summary["kills"]:
		var killer := String(k["killer"])
		if not st.characters.has(killer):
			continue
		var def := BattleData.enemy(String(k["def_id"]))
		var pts := KillReward.points_for_kill(float(def.get("kill_base", 0.0)), float(def.get("threat", 0.0)), rating_power)
		st.kill_points[killer] = float(st.kill_points.get(killer, 0.0)) + pts
		st.kill_counts[killer] = int(st.kill_counts.get(killer, 0)) + 1
		events.append({"type": "kill_recorded", "char": killer, "enemy": String(k["def_id"]), "points": pts})
	for d: Dictionary in summary["deaths"]:
		var victim := String(d["id"])
		var killer := String(d["killer"])
		if st.characters.has(victim) and st.characters.has(killer) and killer != victim and st.characters[victim].is_newbie():
			st.add_ledger(killer, "penalty", float(Rules.get_value("settlement", "kill_newbie")), "杀死新人")
			events.append({"type": "kill_newbie_penalty", "char": killer, "victim": victim})
	for f: String in summary["fled"]:
		if st.is_alive(f):
			_flee(f, events)
	var loc := String(st.pending_battle["location"])
	var won := ["victory", "survived"].has(String(b.outcome))
	var after: Array = st.pending_battle.get("after_victory" if won else "after_defeat", [])
	st.pending_battle = {}
	events.append({"type": "battle_resolved", "outcome": String(b.outcome)})
	Watch.check_wipe(st, events)
	PlotRunner.apply_effects(st, after, loc, events)
	PlotRunner.run_conditional(st, events)
	return _emit(events)


## 道具结算：战斗里用掉的数量先从分队共用的扣，再从自己的扣。
func _settle_items(u: RefCounted, ch: RefCounted, squad_id: String) -> void:
	var shared: Dictionary = st.squads[squad_id]["items"] if squad_id != "" else {}
	var keys: Array = ch.items.keys()
	for k: String in shared:
		if not keys.has(k):
			keys.append(k)
	for k: String in keys:
		var own := float(ch.items.get(k, 0.0))
		var common := float(shared.get(k, 0.0))
		var used := maxf(0.0, own + common - float(u.resources.get(k, own + common)))
		var from_common := minf(used, common)
		if squad_id != "":
			if common - from_common > 0.0:
				shared[k] = common - from_common
			else:
				shared.erase(k)
		var own_left := own - (used - from_common)
		if own_left > 0.0:
			ch.items[k] = own_left
		else:
			ch.items.erase(k)


## 逃离战场：往相邻地点里离绑定对象最远的那个跑（没有绑定就跑第一个相邻地点），可能因此被抹杀。

	elif not st.pending_choice.is_empty():

	elif not st.pending_check.is_empty():
		reason = "check_pending"
	elif not st.squads.has(squad_id) or st.living_members(squad_id).is_empty():
		reason = "no_squad"
	if reason != "":
		events.append({"type": "invalid_action", "reason": reason})
		return false
	return true


## 推进 minutes 分钟：途中到点的定时事件依次发生，每次之后检查条件事件。
## 返回 true 表示时间完整走完；被战斗、选择、说服打断或远征结束返回 false。
func _run_time(minutes: int, events: Array[Dictionary]) -> bool:
	var target: int = st.clock + minutes
	while st.outcome == "ongoing":
		var ev := PlotRunner.next_timed(st)
		if ev.is_empty() or int(ev["at"]) > target:
			break
		Watch.pass_time(st, int(ev["at"]) - st.clock, events)
		if st.outcome != "ongoing":
			return false
		PlotRunner.fire(st, ev, events)
		PlotRunner.run_conditional(st, events)
		if st.is_blocked():
			return false
	Watch.pass_time(st, target - st.clock, events)
	PlotRunner.run_conditional(st, events)
	return st.outcome == "ongoing" and not st.is_blocked()

		"invalid_action":
			return ""
	return ""


static func _name(id: Variant, names: Dictionary) -> String:
	var key := String(id)
	return String(names.get(key, key))


static func _reason(r: String) -> String:
	var table := {
		"enemy_appear": "敌人登场", "darkness": "黑暗", "plot": "剧情", "rest": "休息",
		"tether": "绑定", "turned": "感染", "fight": "战斗", "low_hp": "濒死", "aura": "敌人威压",
	}
	return String(table.get(r, r))


static func _status(id: Variant) -> String:
	var s := String(id)
	var all := BattleData.statuses()
	return String((all[s] as Dictionary).get("name", s)) if all.has(s) else s

	return out


## 同一技能对多个目标时合成一组，界面先选技能再选目标，免得二十个一样的按钮。
func groups() -> Array[Dictionary]:
	var legal: Array[Dictionary] = b.legal_actions()
	var order: Array[String] = []
	var by_key := {}
	for i in legal.size():
		var key := _group_key(legal[i])
		if not by_key.has(key):
			by_key[key] = {"key": key, "label": "", "indices": []}
			order.append(key)
		(by_key[key]["indices"] as Array).append(i)
	var out: Array[Dictionary] = []
	for key in order:
		var g: Dictionary = by_key[key]
		var idx: Array = g["indices"]
		g["label"] = _label(legal[int(idx[0])]) if idx.size() == 1 else "%s（%d 个目标）" % [_skill_name(legal[int(idx[0])]), idx.size()]
		out.append(g)
	return out


func _group_key(a: Dictionary) -> String:
	var t := String(a["type"])
	if t == "skill":
		return "skill:" + String(a["skill"])
	if t == "move":
		return "move:%d" % int(a["row"])
	return t


func _skill_name(a: Dictionary) -> String:
	var skill := String(a.get("skill", ""))
	return String(BattleData.skill(skill).get("name", skill))


func _label(a: Dictionary) -> String:
	var t := String(a["type"])
	match t:
		"skill":
			var title := _skill_name(a)
			var who: Array[String] = []
			for id: String in a.get("targets", []):
				who.append(_target_label(id))
			return "%s → %s" % [title, "、".join(who)] if not who.is_empty() else title
		"move":
			return "换到第 %d 排" % (int(a["row"]) + 1)
		"defend":
			return "防御"
		"gene_lock":
			return "开启基因锁"
		"retreat":
			return "撤退"
	return t


## 只带编号的名字，给意图面板用。
func _short_name(id: String) -> String:
	var name := String(names.get(id, id))
	var tail := id.get_slice("_", id.get_slice_count("_") - 1)
	if tail.is_valid_int():
		name += "#" + tail
	return name


## 目标标签：名字后面加编号和剩余生命，十个丧尸才分得清。
func _target_label(id: String) -> String:
	var name := _short_name(id)
	var tail := id.get_slice("_", id.get_slice_count("_") - 1)
	if tail.is_valid_int():
		name += "#" + tail
	var u: RefCounted = b.field.get_unit(id)
	if u != null:
		name += "（%d）" % int(u.hp)
	return name


func act(index: int) -> Array[Dictionary]:
	var legal: Array[Dictionary] = b.legal_actions()
	if index < 0 or index >= legal.size():
		return []
	return b.act(legal[index])

	var start := maxi(0, b.history.size() - n)
	for i in range(start, b.history.size()):

		if line != "":
			out.append(line)
	return out
