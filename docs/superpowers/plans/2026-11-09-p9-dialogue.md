## 设计要点

- **任务编号用内容哈希**：`line:<npc>:<玩家那句话的 md5 前 10 位>`、`persuade:<npc>:<目标>:<说辞>`、`zhushen:<问题>`。同一句话说第二次命中 `ai_log` 的回放，不重复花钱；回放结果与第一次相同。
- **说服力**：AI 只给 0~3 级和理由，成败仍由 `Expedition.resolve_persuasion` 判（高于难度必成、等于难度一半机会、低于难度失败）。离线时按说辞命中几个 `keywords` 算级（最多 3），没有关键词时超过 20 字算 1 级。
- **门槛仍在规则层**：逼火焰女皇前必须推理出 `queen_controls_lasers`。`Expedition.can_persuade()` 统一回答"现在能不能开口"，界面据此决定是否打开说服框。
- **记忆**：每说一轮记两条（"他对我说：…"和"我说：…"），上限 12 条，超出把最早两条并成一条"更早的事"（与 `persona.gd` 同一口径）。同步回 `persona.memories`，下一轮 `context_text()` 才带得上。
- **信任**：玩家每说一句 +1，0~5。剧本 `trust` 段配门槛（M1：卡普兰/艾丽丝/詹岚 2 点、张杰 3 点），跨过门槛触发一次 `confide`：一句实话 + 剧本给的效果（这里是给线索）。线索进 `intel` 后可能让某条事实从"有线索"变成"可推理"。
- **主神问答**：`data/ai/zhushen_faq.json` 是离线问答表；`forbidden` 里的词（下一部、剧情、原著…）出现在回答里就判不合格，自动换成离线回答；离线表里没有的问题一律"此问题超出你的权限。"
- **离线台词不重复**：人设只有三句兜底台词，连着撞同一句时换成下一句（`Dialogue._vary`），只在 `fallback` 时生效。
- **界面**：对话框盖住整屏，中间 1240×800 的面板；标题栏显示称呼与"在线/离线"，正文是说话人着色的记录，下面一排候选回复（手机上点选，不用打字），再下面是输入框和"说/说服"按钮。远征界面加一行"搭话：…"（当前地点 NPC + 同分队队友，最多四个）；说服区加"自己说（AI 评分）"；主神空间左侧加"问主神"。
- **`dialogue_panel` 用 `Panel` 而不是 `PanelContainer`**：`PanelContainer` 会重排子节点，标题、日志、输入框会挤在一起。

**不在本计划范围**：流式逐字输出（`ai/sse.gd` 已有解析，等需要时再接）、语音、立绘与音效、造物/队友的主动搭话、M2 内容。

```gdscript
extends RefCounted
## 对话层（设计规格 8.1）：把"玩家说了什么"交给 AI 接入层，换成 NPC 的台词或说服力等级。
## 没有密钥、断网、模型答得不合格时，ai/fallback.gd 给一份确定的答案，所以离线也能玩，只是台词固定。
## 记忆与信任存在 store（就是 Session.dialogues），直接随存档走，不需要额外的序列化。
##   store[npc_id] = {"memories": [String], "trust": 0~5}
## 任务 id 用内容哈希而不是递增序号：同一句话说第二次会命中回放，不重复花钱。

const Persona = preload("res://ai/persona.gd")

const TRUST_MAX := 5
const MAX_REPLY_CHOICES := 3


var service: RefCounted
var personas: Dictionary = {}
var store: Dictionary = {}
## 本次会话的对话记录：[{"speaker", "text"}]，不存档（存档只留记忆与信任）。
var log_lines: Array[Dictionary] = []
## 最近一次结果是 "ai"、"fallback" 还是 "replay"，界面拿它显示"在线/离线"。
var last_source: String = ""


func _init(svc: RefCounted, persona_map: Dictionary, save_store: Dictionary) -> void:
	service = svc
	personas = persona_map
	store = save_store


## 这个角色有没有人设（没有就说明它不该由 AI 驱动）。
func has_persona(npc_id: String) -> bool:
	return personas.has(npc_id)


func persona(npc_id: String) -> RefCounted:
	return personas.get(npc_id, null)


func display_name(npc_id: String) -> String:
	var p: RefCounted = persona(npc_id)
	return p.name if p != null else npc_id


func online() -> bool:
	return service != null and service.online()


## 一段能塞进 prompt 的局面摘要；界面按当前地点和时钟拼。
func situation_text(location_name: String, clock: int) -> String:
	return "生化危机一，蜂巢。现在是第 %d 分钟，你在%s。" % [clock, location_name]


## 以 npc 的身份回一句。done 收到 {"say", "source"}。
func say(npc_id: String, line: String, situation: String, done: Callable) -> void:
	var p: RefCounted = persona(npc_id)
	if p == null:
		done.call({"say": "……", "source": "none"})
		return
	var said := line.strip_edges()
	if said != "":
		remember(npc_id, "他对我说：" + said.left(60))
	var task := {
		"id": "line:%s:%s" % [npc_id, said.md5_text().left(10)],
		"kind": "npc_line",
		"tier": "fast",
		"situation": situation,
	}
	service.ask(task, p, func(res: Dictionary) -> void:
		var intent: Dictionary = res["intent"]
		var text := String(intent.get("say", "……"))
		last_source = String(res["source"])
		text = _vary(text, p)
		log_lines.append({"speaker": p.name, "text": text})
		remember(npc_id, "我说：" + text.left(60))
		done.call({"say": text, "source": last_source})
	)


## 给玩家几句可点的回复（手机上比打字省事）。done 收到 {"replies": [String], "source"}。
func suggest(npc_id: String, situation: String, defaults: Array, done: Callable) -> void:
	var p: RefCounted = persona(npc_id)
	var task := {
		"id": "replies:%s:%s" % [npc_id, situation.md5_text().left(10)],
		"kind": "replies",
		"tier": "fast",
		"situation": situation,
		"default_replies": defaults,
	}
	service.ask(task, p, func(res: Dictionary) -> void:
		var intent: Dictionary = res["intent"]
		var list: Array = (intent.get("replies", defaults) as Array).duplicate()
		last_source = String(res["source"])
		done.call({"replies": list.slice(0, MAX_REPLY_CHOICES), "source": last_source})
	)


## 给一段说辞打 0~3 级（引擎再做检定）。离线时按关键词命中数算，见 fallback.gd。
func rate(npc_id: String, argument: String, goal: String, keywords: Array, situation: String, done: Callable) -> void:
	var p: RefCounted = persona(npc_id)
	var task := {
		"id": "persuade:%s:%s:%s" % [npc_id, goal.md5_text().left(6), argument.md5_text().left(8)],
		"kind": "persuade",
		"tier": "fast",
		"situation": situation,
		"argument": argument,
		"goal": goal,
		"keywords": keywords.duplicate(),
	}
	service.ask(task, p, func(res: Dictionary) -> void:
		var intent: Dictionary = res["intent"]
		last_source = String(res["source"])
		remember(npc_id, "我试着说服他：" + argument.left(50))
		done.call({
			"level": clampi(int(intent.get("level", 0)), 0, 3),
			"reason": String(intent.get("reason", "")),
			"source": last_source,
		})
	)


## 问主神。faq 是离线兜底用的问答表，forbidden 是不许说出口的关键词（校验层会拦）。
func ask_zhushen(question: String, faq: Array, forbidden: Array, done: Callable) -> void:
	var task := {
		"id": "zhushen:%s" % question.md5_text().left(12),
		"kind": "zhushen_answer",
		"tier": "fast",
		"situation": "主神空间。轮回者站在巨大的光球下提问。",
		"question": question,
		"faq": faq,
		"forbidden": forbidden,
	}
	service.ask(task, null, func(res: Dictionary) -> void:
		var intent: Dictionary = res["intent"]
		last_source = String(res["source"])
		var answer := String(intent.get("answer", ""))
		if answer == "":
			answer = "此问题超出你的权限。"
		log_lines.append({"speaker": "主神", "text": answer})
		done.call({"answer": answer, "refuse": bool(intent.get("refuse", false)), "source": last_source})
	)


func trust_of(npc_id: String) -> int:
	return int(_entry(npc_id).get("trust", 0))


## 加减信任（0~5）。跨过 def["at"] 门槛时返回一条 confide 事件，带剧本给的效果。
func add_trust(npc_id: String, delta: int, def: Dictionary) -> Array[Dictionary]:
	var before := trust_of(npc_id)
	var after := clampi(before + delta, 0, TRUST_MAX)
	_entry(npc_id)["trust"] = after
	var at := int(def.get("at", TRUST_MAX + 1))
	if before < at and after >= at:
		return [{
			"type": "confide",
			"npc": npc_id,
			"line": String(def.get("line", "")),
			"effects": (def.get("effects", []) as Array).duplicate(true),
		}]
	return []


func memories_of(npc_id: String) -> Array:
	return _entry(npc_id)["memories"]


## 记一条，超过上限时把最早的两条并成一条（规则与 persona.gd 一致）。
func remember(npc_id: String, text: String) -> void:
	if text.strip_edges() == "":
		return
	var list: Array = _entry(npc_id)["memories"]
	list.append(text)
	while list.size() > Persona.MAX_MEMORIES:
		var merged := ("更早的事：%s；%s" % [list[0], list[1]]).left(Persona.MERGED_MAX)
		list.remove_at(0)
		list[0] = merged
	_sync(npc_id)


## 离线人设只有三句，连着重复很出戏：跟上一条一样就换下一句。
func _vary(text: String, p: RefCounted) -> String:
	if last_source != "fallback" or p.fallback_lines.size() < 2 or log_lines.is_empty():
		return text
	var prev: Dictionary = log_lines[log_lines.size() - 1]
	if String(prev["speaker"]) != p.name or String(prev["text"]) != text:
		return text
	var i: int = p.fallback_lines.find(text)
	if i < 0:
		return text
	return p.fallback_lines[(i + 1) % p.fallback_lines.size()]


## 把 store 里的记忆同步回 persona，下一轮 context_text 才看得到。
func _sync(npc_id: String) -> void:
	var p: RefCounted = persona(npc_id)
	if p != null:
		p.memories.assign(_entry(npc_id)["memories"])


func _entry(npc_id: String) -> Dictionary:
	if not store.has(npc_id):
		store[npc_id] = {"memories": [], "trust": 0}
	return store[npc_id]
```

```gdscript
func report_battle(b: RefCounted) -> Array[Dictionary]:
	var events: Array[Dictionary] = []
	if st.pending_battle.is_empty():
		events.append({"type": "invalid_action", "reason": "no_battle"})
		return _emit(events)
	var sid := String(st.pending_battle["squad"])
	var carrier := String(st.pending_battle.get("carrier", ""))
	for u: RefCounted in b.field.units:
		if u.side != "ally" or not st.characters.has(u.character_id):
			continue
		var ch: RefCounted = st.characters[u.character_id]
		UnitFactory.write_back(u, ch)
		_settle_items(u, ch, sid if u.character_id == carrier else "")
		if u.alive and Statuses.has(u, "infection") and not st.infected.has(ch.id):
			st.infected[ch.id] = 0
			events.append({"type": "infected", "char": ch.id})
	var summary := BattleReport.summarize(b.history, b.field.units)
	for k: Dictionary in summary["kills"]:
		var killer := String(k["killer"])
		if not st.characters.has(killer):
			continue
		var def := BattleData.enemy(String(k["def_id"]))
		var pts := KillReward.points_for_kill(float(def.get("kill_base", 0.0)), float(def.get("threat", 0.0)), rating_power)
		st.kill_points[killer] = float(st.kill_points.get(killer, 0.0)) + pts
		st.kill_counts[killer] = int(st.kill_counts.get(killer, 0)) + 1
		events.append({"type": "kill_recorded", "char": killer, "enemy": String(k["def_id"]), "points": pts})
	for d: Dictionary in summary["deaths"]:
		var victim := String(d["id"])
		var killer := String(d["killer"])
		if st.characters.has(victim) and st.characters.has(killer) and killer != victim and st.characters[victim].is_newbie():
			st.add_ledger(killer, "penalty", float(Rules.get_value("settlement", "kill_newbie")), "杀死新人")
			events.append({"type": "kill_newbie_penalty", "char": killer, "victim": victim})
	for f: String in summary["fled"]:
		if st.is_alive(f):
			_flee(f, events)
	var loc := String(st.pending_battle["location"])
	var won := ["victory", "survived"].has(String(b.outcome))
	var after: Array = st.pending_battle.get("after_victory" if won else "after_defeat", [])
	st.pending_battle = {}
	events.append({"type": "battle_resolved", "outcome": String(b.outcome)})
	Watch.check_wipe(st, events)
	PlotRunner.apply_effects(st, after, loc, events)
	PlotRunner.run_conditional(st, events)
	return _emit(events)


## 道具结算：战斗里用掉的数量先从分队共用的扣，再从自己的扣。
func _settle_items(u: RefCounted, ch: RefCounted, squad_id: String) -> void:
	var shared: Dictionary = st.squads[squad_id]["items"] if squad_id != "" else {}
	var keys: Array = ch.items.keys()
	for k: String in shared:
		if not keys.has(k):
			keys.append(k)
	for k: String in keys:
		var own := float(ch.items.get(k, 0.0))
		var common := float(shared.get(k, 0.0))
		var used := maxf(0.0, own + common - float(u.resources.get(k, own + common)))
		var from_common := minf(used, common)
		if squad_id != "":
			if common - from_common > 0.0:
				shared[k] = common - from_common
			else:
				shared.erase(k)
		var own_left := own - (used - from_common)
		if own_left > 0.0:
			ch.items[k] = own_left
		else:
			ch.items.erase(k)


## 逃离战场：往相邻地点里离绑定对象最远的那个跑（没有绑定就跑第一个相邻地点），可能因此被抹杀。
func _flee(char_id: String, events: Array[Dictionary]) -> void:

	if options.is_empty():
		return

			if d > best_d:

				best = n
	st.move_members([char_id], best)
	events.append({"type": "fled_away", "char": char_id, "to": best})
	Watch.check_tether(st, 0, events)

func _can_act(squad_id: String, events: Array[Dictionary]) -> bool:
	var reason := ""
	if st.outcome != "ongoing":
		reason = "ended"
	elif not st.pending_battle.is_empty():
		reason = "battle_pending"
	elif not st.pending_choice.is_empty():
		reason = "choice_pending"
	elif not st.pending_check.is_empty():
		reason = "check_pending"
	elif not st.squads.has(squad_id) or st.living_members(squad_id).is_empty():

## 推进 minutes 分钟：途中到点的定时事件依次发生，每次之后检查条件事件。
## 返回 true 表示时间完整走完；被战斗、选择、说服打断或远征结束返回 false。
func _run_time(minutes: int, events: Array[Dictionary]) -> bool:
	var target: int = st.clock + minutes
	while st.outcome == "ongoing":
		var ev := PlotRunner.next_timed(st)
		if ev.is_empty() or int(ev["at"]) > target:
			break
		Watch.pass_time(st, int(ev["at"]) - st.clock, events)
		if st.outcome != "ongoing":
			return false
		PlotRunner.fire(st, ev, events)
		PlotRunner.run_conditional(st, events)
		if st.is_blocked():
			return false
	Watch.pass_time(st, target - st.clock, events)
	PlotRunner.run_conditional(st, events)
	return st.outcome == "ongoing" and not st.is_blocked()
```

```gdscript
var expedition: RefCounted

func start(chapter: String, seed_value: int = 0) -> bool:
	chapter_path = chapter
	game = GameState.new(seed_value)
	expedition = Campaign.start(game, chapter)
	if expedition == null:
		return false
	phase = "expedition"
	hub = null
	day = 0
	settlement = {}
	dialogues = {}
	return true


func start_m1(seed_value: int = 0) -> bool:
	return start(M1_CHAPTER, seed_value)


func in_expedition() -> bool:
	return phase == "expedition" and expedition != null


func expedition_over() -> bool:
	return expedition != null and String(expedition.st.outcome) != "ongoing"


## 现在为什么不能行动："" 表示可以，否则是 battle / choice / check / ended。
func blocked_reason() -> String:
	if not in_expedition():
		return "ended"
	if expedition.st.outcome != "ongoing":
		return "ended"
	if not expedition.st.pending_battle.is_empty():
		return "battle"
	if not expedition.st.pending_choice.is_empty():
		return "choice"
	if not expedition.st.pending_check.is_empty():
		return "check"
	return ""


## 片子结束（回归、全灭或剧本结局）后结算一次，进入 settlement 阶段。
func settle() -> Dictionary:
	if phase != "expedition" or not expedition_over():
		return {}
	settlement = expedition.finish()
	hub = Hub.new()
	hub.start(game, REST_DAYS)
	phase = "settlement"
	day = hub.day
	return settlement


func enter_hub() -> bool:
	if phase != "settlement":
		return false
	phase = "hub"
	return true


func advance_day() -> bool:
	if phase != "hub" or hub == null or hub.can_depart():
		return false
	hub.rest(1)
	day = hub.day
	return true


func can_depart() -> bool:
	return phase == "hub" and hub != null and hub.can_depart()


## 走光柱出发；M2 的内容还没做，先把阶段停在 done。
func depart() -> bool:
	if not can_depart():
		return false
	phase = "done"
	return true


func to_dict() -> Dictionary:
	return {
		"version": VERSION,
		"phase": phase,
		"chapter": chapter_path,
		"day": day,
		"game": game.to_dict() if game != null else {},
		"expedition": expedition.to_dict() if expedition != null else {},
		"hub": hub.to_dict() if hub != null else {},
		"settlement": settlement.duplicate(true),
		"dialogues": dialogues.duplicate(true),
	}


## 读档失败返回 false（版本不对、字段缺失、剧本读不到）。
func load_dict(d: Dictionary) -> bool:
	if int(d.get("version", 0)) != VERSION or not d.has("game") or not d.has("chapter"):
		return false
	chapter_path = String(d["chapter"])
	var chapter := Campaign.load_chapter(chapter_path)
	if chapter.is_empty():
		return false
	game = GameState.new()
	game.load_dict(d["game"])
	phase = String(d.get("phase", "expedition"))
	if not PHASES.has(phase):
		return false
	day = int(d.get("day", 0))
	settlement = (d.get("settlement", {}) as Dictionary).duplicate(true)
	dialogues = (d.get("dialogues", {}) as Dictionary).duplicate(true)
	var h: Dictionary = d.get("hub", {})
	if not h.is_empty():
		hub = Hub.new()
		hub.load_dict(h, game)
		day = hub.day
	var e: Dictionary = d.get("expedition", {})
	if e.is_empty():
		expedition = null
		return phase != "expedition"
	var sc := Scenario.load_file(String(chapter["scenario"]))
	if sc == null:
		return false
	expedition = Expedition.new()
	expedition.load_dict(e, game, sc)
	return true


func save_to(path: String) -> Error:
	return SaveService.write_dict(path, to_dict())


static func load_from(path: String) -> RefCounted:
	var d := SaveService.read_dict(path)
	if d.is_empty():
		return null
	var s := new()
	if not s.load_dict(d):
		return null
	return s
```

```gdscript
		if panel != p:
			return

func _on_said(text: String, npc_id: String) -> void:
	var p: Control = panel
	if p == null:
		return
	p.push("我", text)
	p.set_status("……")
	p.set_replies([])
	dialogue.say(npc_id, text, _situation(), func(res: Dictionary) -> void:
		if panel != p:
			return
		p.push(dialogue.display_name(npc_id), String(res["say"]))
		p.set_status(_source_text(String(res["source"])))
		_after_said(npc_id, p)
	)


## 说过话就涨信任；到剧本给的门槛，对方会说一句实话，可能给线索。
func _after_said(npc_id: String, p: Control) -> void:
	var screen := _expedition_screen()
	if screen == null:
		return
	var def: Dictionary = screen.view.trust_def(npc_id)
	if def.is_empty():
		return
	for ev: Dictionary in dialogue.add_trust(npc_id, 1, def):
		if String(ev.get("line", "")) != "":
			p.push(dialogue.display_name(npc_id), String(ev["line"]))
		_push_events(session.expedition.apply_effects(ev["effects"], _location_id()), p)
	screen.refresh()


func _push_events(events: Array, p: Control) -> void:
	var names := _event_names()
	for e: Dictionary in events:
		var text := EventText.describe(e, names)
		if text != "":
			p.push("（结果）", text)


## 事件文本要的不只是人名：地点、剧情事件、线索都得有中文名，不然日志里全是 id。
func _event_names() -> Dictionary:
	var out := _names()
	if session.expedition == null:
		return out
	var sc: RefCounted = session.expedition.st.scenario
	for loc_id: String in sc.data.get("locations", {}):
		out[loc_id] = String(sc.location(loc_id).get("name", loc_id))
	for ev: Dictionary in sc.events():
		out[String(ev["id"])] = String(ev.get("label", ev["id"]))
	for npc_id: String in sc.data.get("npcs", {}):
		out[npc_id] = String(sc.data["npcs"][npc_id].get("name", npc_id))
	for fact_id: String in sc.data.get("facts", {}):
		var f: Dictionary = sc.data["facts"][fact_id]
		var statement := String(f.get("statement", fact_id))
		out[fact_id] = statement
		for clue_id: String in f.get("clues", {}):
			out[clue_id] = statement
	return out


func _on_persuade_free() -> void:
```

```gdscript
func _found_text(node: Node, needle: String) -> bool:
	if node is Label and String(node.text).contains(needle):
		return true
	if node is Button and String(node.text).contains(needle):
		return true
	for c in node.get_children():
		if _found_text(c, needle):
			return true
	return false


func _find_button(node: Node, text: String) -> Button:
	if node is Button and String(node.text) == text:
		return node
```

## 已知风险与缓解

- **在线路径没有真机验证**：没有密钥，所有测试与验收只覆盖 `fallback` 与 `replay` 两条路。在线路径的请求组装、解析、校验已在 P6 测试过；真机联调等密钥到位后单独做。

- **信任只增不减**：说错话不扣信任。等有"说错话"的判定（例如 AI 给 0 级且话题敏感）再补。
- **对话不耗时**：搭话不推进时钟，玩家可以站在安全地点把所有人聊完。M1 里可接受；若以后要压时间，可给每次搭话加 1~2 分钟。
