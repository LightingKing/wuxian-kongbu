## 设计要点

- **为什么合成**：仓库不放二进制资源；8 个短音效用配方（时长、起始/结束频率、噪声比、衰减、音量）就能覆盖，体积为零。
- **谁负责出声**：`Sfx` 的实例是宿主（`main` 的 `_ready` 里 `add_child(Sfx.new())`），静态 `play()` 检查宿主是否有效、是否在树里、音频驱动是否 Dummy，然后转给实例。
- **为什么绕开 Dummy**：无头跑测试时 Dummy 驱动不会结束播放，`AudioStreamPlayer` 会一直 `playing`，池子被占满还留下退出警告。直接不播，测试环境干净。
- **事件驱动**：`play_for_events(events)` 从规则层返回的事件里挑声音——`action`（技能带 bullet 标签）→ 枪声，`damage` → 命中，`death` → 死亡，`miss`/`fled` → 取消，`exchange_done` 等 → 金币，`*_refused` → 取消。战斗与主神空间共用同一张映射表。
- **按钮点击**：`UiTheme.button()` 里统一接 `click`，所有按钮一次覆盖；toggle 类按钮也一样响。
- **音量**：走 Master 总线，设置页的"主音量"直接作用于全部音效。

**不在本计划范围**：背景音乐、环境音、语音、3D 定位音、音频总线分轨（音乐/音效各一条）、音效随机变调。

```gdscript
extends Node
## 音效：运行时用正弦波 + 白噪声合成，不依赖音频文件，也不进仓库。
## 用法：main 里 add_child(Sfx.new()) 设好宿主，其余地方 Sfx.play("click")。
## 没有宿主时（无头测试直接调界面）静默返回，不报错、不建节点。
## 音色是刻意做旧的芯片味：短、干、不抢戏，够让点击和开枪有反馈就行。

const BattleData = preload("res://core/battle/battle_data.gd")

const RATE := 22050
const POOL := 8

## dur 秒、f0→f1 频率滑音、noise 噪声占比、decay 指数衰减、vol 音量。
const RECIPES := {
	"click": {"dur": 0.05, "f0": 900.0, "f1": 900.0, "noise": 0.0, "decay": 0.015, "vol": 0.30},
	"gunshot": {"dur": 0.22, "f0": 190.0, "f1": 55.0, "noise": 0.72, "decay": 0.05, "vol": 0.70},
	"hit": {"dur": 0.12, "f0": 150.0, "f1": 70.0, "noise": 0.5, "decay": 0.03, "vol": 0.50},
	"hurt": {"dur": 0.26, "f0": 480.0, "f1": 180.0, "noise": 0.12, "decay": 0.09, "vol": 0.42},
	"death": {"dur": 0.48, "f0": 210.0, "f1": 50.0, "noise": 0.28, "decay": 0.18, "vol": 0.55},
	"coin": {"dur": 0.14, "f0": 1320.0, "f1": 1760.0, "noise": 0.0, "decay": 0.05, "vol": 0.32},
	"confirm": {"dur": 0.18, "f0": 660.0, "f1": 990.0, "noise": 0.0, "decay": 0.07, "vol": 0.32},
	"cancel": {"dur": 0.16, "f0": 420.0, "f1": 280.0, "noise": 0.0, "decay": 0.06, "vol": 0.32},
}

static var host: Node = null
## 缓存与播放器挂在实例上（不是 static），这样随节点释放，不会在退出时被算成泄漏。
var _cache: Dictionary = {}
var _players: Array[AudioStreamPlayer] = []


func _ready() -> void:
	host = self


func _exit_tree() -> void:
	if host == self:
		host = null


## 播一个音效；没有宿主、id 不认识都只是安静地不做。
static func play(id: String) -> void:
	if host == null or not is_instance_valid(host) or not host.is_inside_tree():
		return
	# 无头跑测试时音频驱动是 Dummy：播放不会结束，播放器会一直占着，干脆不播。
	if AudioServer.get_driver_name() == "Dummy":
		return
	host._play(id)


func _play(id: String) -> void:
	var stream := _wav(id)
	if stream == null:
		return
	var p := _pick()
	p.stream = stream
	p.play()


func _pick() -> AudioStreamPlayer:
	for p in _players:
		if is_instance_valid(p) and not p.playing:
			return p
	if _players.size() >= POOL:
		return _players[0]
	var p := AudioStreamPlayer.new()
	host.add_child(p)
	_players.append(p)
	return p


## 同一个 id 只合成一次。
func _wav(id: String) -> AudioStreamWAV:
	if _cache.has(id):
		return _cache[id]
	if not RECIPES.has(id):
		return null
	var r: Dictionary = RECIPES[id]
	var stream := _synth(float(r["dur"]), float(r["f0"]), float(r["f1"]), float(r["noise"]), float(r["decay"]), float(r["vol"]))
	_cache[id] = stream
	return stream


static func _synth(dur: float, f0: float, f1: float, noise: float, decay: float, vol: float) -> AudioStreamWAV:
	var n := maxi(int(dur * RATE), 1)
	var bytes := PackedByteArray()
	bytes.resize(n * 2)
	var rng := RandomNumberGenerator.new()
	rng.seed = 20261109
	for i in n:
		var t := float(i) / float(RATE)
		var k := float(i) / float(maxi(n - 1, 1))
		var f := lerpf(f0, f1, k)
		var env := exp(-t / maxf(decay, 0.001))
		var wave := sin(TAU * f * t) * (1.0 - noise) + rng.randf_range(-1.0, 1.0) * noise
		bytes.encode_s16(i * 2, int(clampf(wave * env * vol, -1.0, 1.0) * 32767.0))
	var w := AudioStreamWAV.new()
	w.format = AudioStreamWAV.FORMAT_16_BITS
	w.mix_rate = RATE
	w.stereo = false
	w.data = bytes
	return w


## 给界面用：从一批事件里挑该响的声音（战斗与主神空间共用）。
static func play_for_events(events: Array) -> void:
	for e: Dictionary in events:
		match String(e.get("type", "")):
			"action":
				var act: Dictionary = e.get("action", {})
				if String(act.get("type", "")) == "skill":
					var all := BattleData.skills()
					var skill_id := String(act.get("skill", ""))
					if all.has(skill_id) and ((all[skill_id] as Dictionary).get("tags", []) as Array).has("bullet"):
						play("gunshot")
			"damage":
				play("hit")
			"death":
				play("death")
			"miss":
				play("cancel")
			"fled":
				play("cancel")
			"exchange_done", "attribute_bought", "repaired", "human_created", "trained", "ticket_split":
				play("coin")
			"exchange_refused", "hub_refused":
				play("cancel")
```

## Esc：标题画面先关覆盖层；游戏里先关对话框，再关暂停菜单，最后打开暂停菜单。
func _unhandled_input(event: InputEvent) -> void:
	var k := event as InputEventKey
	if k == null or not k.pressed or k.echo or k.keycode != KEY_ESCAPE:
		return
	if title_screen != null and is_instance_valid(title_screen):
		if title_screen.handle_escape():
			accept_event()
		return
	if panel != null and is_instance_valid(panel):
		panel.press_close()
		accept_event()
		return
	if pause != null:
		_close_pause()
	else:
		_open_pause()
	accept_event()

func _on_talk(npc_id: String) -> void:
	var title: String = dialogue.display_name(npc_id)
	_open_panel(title, "%s · 第 %d 分钟" % [_location_name(), int(session.expedition.st.clock)], "talk")
	var p: Control = panel
	p.set_portrait(npc_id)
	p.set_status("……")

	if p == null:
		return
	p.push("我", text)
	p.set_status("……")
	p.set_replies([])

		if panel != p:
			return
		p.push(dialogue.display_name(npc_id), String(res["say"]))
		p.set_status(_source_text(String(res["source"])))
		_after_said(npc_id, p)
	)


## 说过话就涨信任；到剧本给的门槛，对方会说一句实话，可能给线索。
func _after_said(npc_id: String, p: Control) -> void:
	var screen := _expedition_screen()
	if screen == null:
		return
	var def: Dictionary = screen.view.trust_def(npc_id)
	if def.is_empty():
		return
	for ev: Dictionary in dialogue.add_trust(npc_id, 1, def):
		if String(ev.get("line", "")) != "":
			p.push(dialogue.display_name(npc_id), String(ev["line"]))
		_push_events(session.expedition.apply_effects(ev["effects"], _location_id()), p)
	screen.refresh()


func _push_events(events: Array, p: Control) -> void:
	var names := _event_names()
	for e: Dictionary in events:
		var text := EventText.describe(e, names)
		if text != "":
			p.push("（结果）", text)


## 事件文本要的不只是人名：地点、剧情事件、线索都得有中文名，不然日志里全是 id。
func _event_names() -> Dictionary:
	var out := _names()
	if session.expedition == null:
		return out
	var sc: RefCounted = session.expedition.st.scenario
	for loc_id: String in sc.data.get("locations", {}):
		out[loc_id] = String(sc.location(loc_id).get("name", loc_id))
	for ev: Dictionary in sc.events():
		out[String(ev["id"])] = String(ev.get("label", ev["id"]))
	for npc_id: String in sc.data.get("npcs", {}):
		out[npc_id] = String(sc.data["npcs"][npc_id].get("name", npc_id))
	for fact_id: String in sc.data.get("facts", {}):
		var f: Dictionary = sc.data["facts"][fact_id]
		var statement := String(f.get("statement", fact_id))
		out[fact_id] = statement
		for clue_id: String in f.get("clues", {}):
			out[clue_id] = statement
	return out


func _on_persuade_free() -> void:
	var screen := _expedition_screen()
	if screen == null:
		return
	var check: Dictionary = session.expedition.st.pending_check
	if check.is_empty():
		return
	if not bool(session.expedition.can_persuade(String(screen.view.best_thinker_id()))["ok"]):
		return
	var npc := String(check["npc"])
	_open_panel("%s · 说服" % dialogue.display_name(npc), String(check["goal"]), "persuade")
	panel.set_portrait(npc)
	panel.push("（局面）", "要让对方点头：%s" % String(check["goal"]))
	panel.persuading.connect(_on_persuade_text)


func _on_persuade_text(text: String) -> void:
	var p: Control = panel
	var screen := _expedition_screen()
	if p == null or screen == null:
		return
	var check: Dictionary = session.expedition.st.pending_check
	if check.is_empty():
		p.push("（提示）", "现在没有要说服的人。")
		return
	var npc := String(check["npc"])
	p.push("我", text)
	p.set_status("……")
	dialogue.rate(npc, text, String(check["goal"]), check.get("keywords", []), _situation(), func(res: Dictionary) -> void:
		if panel != p:
			return
		p.push("（判定）", "说服力 %d/3。%s" % [int(res["level"]), String(res["reason"])])
		_push_events(session.expedition.resolve_persuasion(int(res["level"])), p)
		p.set_status(_source_text(String(res["source"])))
		screen.refresh()
	)


func _on_ask_zhushen() -> void:
	_open_panel("主神", "巨大的光球悬在头顶，没有表情。", "talk")
	panel.set_portrait("zhushen")
	panel.push("主神", "问。")
	panel.submitted.connect(_on_zhushen_question)


func _on_zhushen_question(text: String) -> void:

	if p == null:
		return
	p.push("我", text)
	p.set_status("……")

		if panel != p:
			return
		p.push("主神", String(res["answer"]))
		p.set_status(_source_text(String(res["source"])))
	)


func _zhushen_faq() -> Array:
	var res := DataLoader.load_json(ZHSHEN_FAQ)
	if not res.ok:
		return []
	return (res.data as Dictionary).get("faq", [])

## 已知风险与缓解

- **芯片味太足**：正弦 + 噪声做出来的声音偏"8-bit"，跟日系插画的调性不完全搭。等有了 BGM 和正式音效，这套只留作 UI 反馈。
- **无头听不到声音**：测试只验证波形与调用链，验证不了"好不好听"。真机点一遍算验收。
- **枪声只有一种**：沙漠之鹰和手枪共用 `gunshot`。以后按技能分档再拆。
- **同时发声上限 8 个**：超过就复用第 0 个（会截断）。正常操作不会触发。
- **主音量改的是 Master**：等拆了音乐/音效总线，这里要改成只调 SFX 总线。
